{{/* Returns the URL of an image. .name file name, path under assets/ or static/, or a full URL .page page bundle searched first (optional) .width downscale raster images wider than this (optional) */}} {{- $url := "" }} {{- $name := .name }} {{- with $name }} {{- $img := false }} {{- with $.page }}{{ $img = .Resources.Get $name }}{{ end }} {{- if not $img }}{{ $img = resources.Get $name }}{{ end }} {{- if and $img (ne $img.MediaType.MainType "image") }} {{- errorf "%q is not an image (a cover folder must end with \"/\")" $name }} {{- end }} {{- with $img }} {{- if and $.width (reflect.IsImageResourceProcessable .) (gt .Width $.width) }} {{- $img = .Process (printf "resize %dx" $.width) }} {{- end }} {{- $url = $img.RelPermalink }} {{- else }} {{- $url = $name | absURL }} {{- end }} {{- end }} {{- return $url }}
